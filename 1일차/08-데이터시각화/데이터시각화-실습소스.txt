fig, (ax1, ax2) = plt.subplots(1, 2, constrained_layout=True)  # 1행 2열로 구성된 서브플롯 생성합니다.

sns.lineplot(
    data=df, x="연", y="매출액", errorbar=None, marker="o", ax=ax1, color="r"
)  # ax1에 선 그래프를 그립니다. errorbar=None은 그대로 두셔도 됩니다.
ax1.set_title("연도별 매출액")  # ax1의 제목을 설정합니다.

sns.barplot(
    df, x="연", y="매출액", hue="공휴일", errorbar=None, ax=ax2
)  # ax2에 막대 그래프를 그립니다.
# "공휴일" 값을 기준으로 그룹을 나눕니다.
ax2.set_title("공휴일 여부에 따른 매출액")  # ax2의 제목을 설정합니다.

plt.show()

**실습4

histtype 부분은 다음 코드가 좋음
data1 = df_sat["어린이"]
data2 = df_sat["청소년"]

plt.figure()
plt.hist([data1, data2], histtype="bar", label=["어린이", "청소년"])
plt.title("bar")
plt.legend()
plt.show()

plt.figure()
plt.hist([data1, data2], histtype="barstacked", label=["어린이", "청소년"])
plt.title("barstacked")
plt.legend()
plt.show()

plt.figure()
plt.hist([data1, data2], histtype="step", label=["어린이", "청소년"])
plt.title("step")
plt.legend()
plt.show()

plt.figure()
plt.hist([data1, data2], histtype="stepfilled", label=["어린이", "청소년"])
plt.title("stepfilled")
plt.legend()
plt.show()
