from elice_utils import EliceUtils

import pandas as pd

a = pd.Series([20, 15, 30, 25, 35], name='age')
b = pd.Series([68.5, 60.3, 53.4, 74.1, 80.7], name='weight')
c = pd.Series([180, 165, 155, 178, 185], name ='height')

## 1. human DataFrame 만들기
# human = pd.DataFrame(None)
human = pd.DataFrame([a, b, c])
print(human,'\n')

def main():
    ### 2. loc(), iloc() 함수를 이용하여 `age`를 추출
    print(human.loc['age'],'\n')
    print(human.iloc[0],'\n')
    
    ### 3. loc(), iloc() 함수를 이용하여 `weight`와 `height`만 추출
    print(human.loc['weight':'height'],'\n')
    # print(human.loc[['weight','height']],'\n')
    print(human.iloc[1:],'\n')
     
    sex = ['F','M','F','M','F']
    ### 4.새로운 데이터 `sex`를 `human`에 추가
    human.loc['sex'] = sex
    print(human,'\n')
    
    ### 5. `human`에서 `height`를 삭제
    tmp = human.drop('height', axis=0)
    print(tmp,'\n')


if __name__ == "__main__":
    main()

import numpy as np
import pandas as pd

A = pd.DataFrame(np.random.randint(0, 10, (2, 2)))
B = pd.DataFrame(np.random.randint(0, 10, (3, 3)))

print("DataFrame A \n", A)
print("DataFrame B \n", B)

# 함수를 이용해서 연산을 해보세요.
# NaN 값이 생기지 않도록 fill_value에 값을 넣어보세요
# A + B
add = A.add(B, fill_value=0.0)
print(add,'\n')
# A - B
sub = A.sub(B, fill_value=0.0)
print(sub,'\n')
# A * B
mul = A.mul(B, fill_value=0.0)
print(mul,'\n')
# A / B
div = A.div(B, fill_value=0.0)
print(div,'\n')


# 3 x 3 데이터프레임을 정렬해보세요.
C = pd.DataFrame([[1,3,5],[15,10,5],[2,8,5]], index = ['a','b','c'], columns = ['d','e','f'])
print(C)

# c 행을 기준으로 열을 오름차순 정렬
row_C = C.sort_values(by='c', ascending=True, axis=1)

# e 열을 기준으로 행을 내림차순 정렬
column_C = C.sort_values(by='e', ascending=False, axis=0)

print(row_C,'\n')
print(column_C,'\n')

# 데이터 csv로 저장 및 불러오기
# index를 False로 설정하면 저장할 때 추가 인덱스를 달지 않습니다.

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

from elice_utils import EliceUtils
elice_utils = EliceUtils()

def main():
    x = np.arange(10)
    
    # 초기 figure와 축을 설정합니다.
    fig, ax = plt.subplots()
    
    # y = x 그래프를 그립니다. 따라서 x 데이터는 x, y 데이터도 x로 설정합니다.
    # None
    # label은 'y=x'로 설정하고, 마커는 'o', 마커 색깔은 'blue', 그래프의 선 스타일은 ':'로 설정합니다.
    ax.plot(x, x, marker='o', color='blue', linestyle=':')
    
    # y = x^2 그래프를 그립니다. 따라서 x 데이터는 x, y 데이터는 x**2으로 설정합니다.
    # label은 'y=x^2'로 설정하고, 마커는 '^', 마커 색깔은 'red', 그래프의 선 스타일은 '--'로 설정합니다.
    # None
    ax.plot(x, x**2, marker='^', color='red', linestyle='--')
    
    # 그래프 제목을 'Graph'로 설정합니다.
    # None
    ax.set_title("Graph")
    
    # x label은 'x', y label은 'y'로 설정합니다.
    # None
    ax.set_xlabel("x")
    ax.set_ylabel("y")
    
    # x 범위는 0부터 10까지, y 범위는 0부터 100까지로 설정합니다.
    # None
    ax.set_xlim(0, 10)
    ax.set_ylim(0, 100)
    
    # 범례의 위치는 'upper left'로 하고, 그림자 효과는 넣고, 테두리는 둥글게 합니다.
    # None
    ax.legend(loc='upper left', shadow=True, fancybox=True)
        
    # figure를 "plot.png"라는 이름으로 저장하세요.
    # None
    fig.savefig("plot.png")
    
    # 엘리스 플랫폼에서 그래프를 출력하기 위한 코드입니다.
    elice_utils.send_image("plot.png")

if __name__ == "__main__":
    main()

import numpy as np
import matplotlib.pyplot as plt
import csv

from elice_utils import EliceUtils
elice_utils = EliceUtils()

def main():
    
    data_x = []
    data_y = []
    
    with open('./data/data.csv', newline='') as f:
        reader = csv.reader(f)
        for row in reader:
            data_x.append(float(row[0]))
            data_y.append(float(row[1]))
    
    # 2x2개의 그래프를 그릴 수 있는 초기 figure와 축을 설정합니다.
    fig, axes = plt.subplots(2, 2)
    
    """
    Scatter 그래프 그리기
    """
    
    colors = np.random.randint(0,100,500)
    
    # figure의 (0,0) 위치에 scatter 그래프를 그립니다.
    # x 데이터는 data_x, y 데이터는 data_y, 데이터 포인트의 색깔은 colors, 사이즈는 2, 투명도는 0.7로 설정합니다.
    # None
    axes[0,0].scatter(data_x, data_y, c=colors, s=2, alpha=0.7)
    
    """
    Bar 그래프 그리기
    """
    
    bar_x = np.arange(10)
    
    # figure의 (0,1) 위치에 Bar 그래프를 그립니다.
    # x 데이터는 bar_x, y 데이터는 bar_x**2로 설정합니다.
    # None
    axes[0,1].bar(bar_x, bar_x**2)
    
    """
    Multi-Bar 그래프 그리기
    """
    
    x = np.array([3,2,1])
    y = np.array([2,3,2])
    z = np.array([1,3,4])
    data1 =  [x, y, z]

    x_ax =  np.arange(3)
    
    for i in x_ax:
        # figure의 (1,0) 위치에 Bar 그래프를 그립니다.
        # x 데이터는 x_ax, y 데이터는 각각 x,y,z로 설정합니다.
        # axes[None].bar(None, None, bottom=np.sum(data1[:i], axis=0))
        axes[1,0].bar(x_ax, data1[i], bottom=np.sum(data1[:i], axis=0))
        
    # figure의 (1,0) 위치에서 x축 데이터를 병렬적으로 설정합니다.
    # None
    axes[1,0].set_xticks(x_ax)
    
    # figure의 (1,0) 위치에서 x축 label을 'A', 'B', 'C'로 설정합니다.
    # None
    axes[1,0].set_xticklabels(['A', 'B', 'C'])
    
    """
    Histogram 그래프 그리기
    """
    
    data = np.array(data_x)
    
    # figure의 (1,1) 위치에 Histogram 그래프를 그립니다.
    # 입력될 데이터는 data, Histogram 표현시 분할되는 개수는 50으로 설정합니다.
    # None
    axes[1,1].hist(data, bins=50)
    
    # figure를 저장하고 엘리스 플랫폼에서 그래프를 출력합니다.
    fig.savefig("plot.png")
    elice_utils.send_image("plot.png")

if __name__ == '__main__':
    main()
