**실습 1
 
 ## 매개변수 a와 b의 곱을 반환하는 함수 multi를 선언합니다.
def multi(a, b):
    return a * b

## 매개변수 a를 b로 나눈 몫을 반환하는 함수 div를 선언합니다.
def div(a, b):
    return a // b


## 정수 두 개를 한 줄에 한 개씩 입력받습니다.
num1 = int(input())
num2 = int(input())

## multi, div 함수를 순서대로 호출하여 연산한 결과를 출력합니다.
print(multi(num1, num2))
print(div(num1, num2))
===========================================================================================================

**실습 2
## 매개변수 a와 b의 곱과, 나눈 몫을 한 번에 반환하는 함수 cal을 선언합니다.
def cal(a, b):
    return a*b, a // b


## 정수 두 개를 한 줄에 한 개씩 입력받습니다.
num1 = int(input())
num2 = int(input())

## cal 함수를 호출하여 변수 multi, div에 각각 연산한 결과를 저장합니다.
multi, div = cal(num1, num2)

print(multi)
print(div)

**실습4
## 요소를 제곱하는 람다 함수를 정의하여 변수 square에 저장하세요.
square = lambda x: x**2

# 정수를 공백을 기준으로 입력받아 리스트로 변환하세요.
numbers = list(map(int, input().split()))

# map 함수를 사용하여 리스트의 각 요소에 람다 함수를 적용하세요.
squared_numbers = list(map(square, numbers))
print(squared_numbers)
