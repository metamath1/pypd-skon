**실습10
input_num = int(input('숫자를 입력해 주세요 : '))
print(input_num,'을(를) 입력받았습니다.')

# 1번을 해보세요!
                 # if 조건 : 결과
# 2번을 해보세요!
                 # elif 조건 : 결과
# 3번을 해보세요!
                 # else : 결과
if input_num > 0:
    print(input_num, "는(은) 양수입니다!")
elif input_num < 0:
    print(input_num, "는(은) 음수입니다!")
else:
    print(input_num, "는(은) 0입니다!")
